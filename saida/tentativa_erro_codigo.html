#include <stdio.h>
#include <stdlib.h>
#define MAXN 20

int col[MAXN];   /* col[lin] = coluna da rainha da linha lin */
int total = 0;

/* devolve 1 se a rainha em (lin, c) não é atacada pelas anteriores */
int pode(int lin, int c) {
  int i;
  for (i = 0; i < lin; i++) {
    if (col[i] == c) return 0;                 /* mesma coluna */
    if (abs(col[i] - c) == lin - i) return 0;  /* mesma diagonal */
  }
  return 1;
}

/* tentativa e erro: tenta todas as colunas da linha lin */
void rainhas(int lin, int n) {
  int c, i;
  if (lin == n) {                /* solução completa encontrada */
    total++;
    if (total == 1) {
      printf("Primeira solucao:\n");
      for (i = 0; i < n; i++) printf("linha %d -> coluna %d\n", i, col[i]);
    }
    return;
  }
  for (c = 0; c < n; c++) {
    if (pode(lin, c)) {
      col[lin] = c;              /* tenta */
      rainhas(lin + 1, n);       /* avança; ao retornar, volta atrás */
    }
  }
}

int main(void) {
  int n = 8;
  rainhas(0, n);
  printf("Total de solucoes para N=%d: %d\n", n, total);
  return 0;
}

#include <stdio.h>

long tentativas;

void troca(int *a, int *b) { int t = *a; *a = *b; *b = t; }

/* tenta cada elemento restante na posição k; se poda != 0, descarta logo
   um candidato menor que v[k-1], pois nenhuma continuação ficaria ordenada */
int ordena_te(int v[], int n, int k, int poda) {
  int i;
  if (k == n) {                                  /* todas as posições preenchidas */
    for (i = 1; i < n; i++) if (v[i] < v[i - 1]) return 0;
    return 1;
  }
  for (i = k; i < n; i++) {
    if (poda && k > 0 && v[i] < v[k - 1]) continue;
    tentativas++;
    troca(&v[k], &v[i]);                         /* tenta v[i] na posição k */
    if (ordena_te(v, n, k + 1, poda)) return 1;
    troca(&v[k], &v[i]);                         /* não deu: desfaz (volta atrás) */
  }
  return 0;
}

void teste(const int orig[], int n, int poda) {
  int v[16], i;
  for (i = 0; i < n; i++) v[i] = orig[i];
  tentativas = 0;
  ordena_te(v, n, 0, poda);
  printf("  %s: %9ld tentativas ->", poda ? "com poda" : "sem poda", tentativas);
  for (i = 0; i < n; i++) printf(" %d", v[i]);
  printf("\n");
}

int main(void) {
  int a[] = {3, 1, 4, 2};
  int b[] = {19, 2, 3, 7, 10, 8, 5, 4, 1, 15};
  printf("n = 4\n");  teste(a, 4, 0);  teste(a, 4, 1);
  printf("n = 10\n"); teste(b, 10, 0); teste(b, 10, 1);
  return 0;
}
